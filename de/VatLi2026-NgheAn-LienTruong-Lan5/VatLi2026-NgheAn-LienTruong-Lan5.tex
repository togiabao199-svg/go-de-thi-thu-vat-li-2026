		{\True Việc các điểm thực nghiệm nằm gần trên một đường thẳng cho thấy số liệu đo được phù hợp với mô hình phân rã phóng xạ theo hàm mũ}
		{\True Nếu nhà máy thay thế nguồn ${}^{60}\text{Co}$ bằng một đồng vị phóng xạ khác có chu kì bán rã ngắn hơn, đồ thị biểu diễn mối liên hệ giữa $\ln H$ và $t$ của nguồn mới sẽ có độ dốc lớn hơn}
		{Giả sử tại thời điểm $t=0$, một lô thanh long cần được chiếu xạ bằng nguồn ${}^{60}\text{Co}$ mới trong thời gian $15$ phút để đạt liều chiếu xạ theo quy chuẩn. Đến ngày thứ $750$ do nguồn phóng xạ bị suy giảm, để đạt cùng liều lượng chiếu xạ chuẩn đó cho một lô thanh long tương tự thì thời gian chiếu cần phải kéo dài $25$ phút}
		\loigiai{
			$H=H_0e^{-\lambda t}\Rightarrow \ln H=\ln H_0-\lambda t$: đồ thị $\ln H$ theo $t$ là đường thẳng có hệ số góc $-\lambda$.
			\begin{itemchoice}
				\itemch \textbf{Sai.} Tia $\gamma$ là sóng điện từ (dòng photon), không mang điện.
				\itemch \textbf{Đúng.} $\ln H$ phụ thuộc bậc nhất vào $t$ đúng với quy luật phân rã theo hàm mũ.
				\itemch \textbf{Đúng.} $\lambda=\dfrac{\ln 2}{T}$: $T$ nhỏ hơn thì $\lambda$ lớn hơn, đồ thị dốc hơn.
				\itemch \textbf{Sai.} $\lambda=\dfrac{0{,}624-0{,}600}{160-90}\approx 3{,}43\cdot 10^{-4}$~ngày$^{-1}$. Cùng liều: $H_0\cdot 15=H t\Rightarrow t=15e^{\lambda\cdot 750}\approx 19{,}4$~phút.
			\end{itemchoice}
		}

		\label{NgheAn-LienTruong-Lan5-III1}\textit{Dùng thông tin sau cho câu~\ref{NgheAn-LienTruong-Lan5-III1} và câu~\ref{NgheAn-LienTruong-Lan5-III2}:} Có hai cái ống cùng nối chung vào một vòi trộn. Mỗi ống có một cái van để điều chỉnh lưu lượng nước trong ống từ $0{,}00$ đến giá trị cực đại $J_0=1{,}00$~lít/s. Trong các ống 1 và 2, nước chảy ra với nhiệt độ $t_1=11\,^\circ\text{C}$ và $t_2=55\,^\circ\text{C}$. Bỏ qua nhiệt lượng mất mát.\\
		Để nước chảy ra khỏi vòi trộn đạt nhiệt độ $44\,^\circ\text{C}$, cần điều chỉnh các van để tỉ lệ lưu lượng nước trong ống 2 so với ống 1 bằng bao nhiêu (làm tròn kết quả đến chữ số hàng phần mười)?
		\par
		\shortans[oly]{3{,}0}
		\loigiai{
			Cân bằng nhiệt (trong mỗi giây): $J_1(t-t_1)=J_2(t_2-t)\Rightarrow \dfrac{J_2}{J_1}=\dfrac{44-11}{55-44}=3{,}0$.
		}
	\end{ex}

	\NeedspaceCau{3.5\baselineskip}
	\begin{ex}%[VL12-C1-VD]
		\label{NgheAn-LienTruong-Lan5-III2}Lưu lượng nước cực đại chảy khỏi vòi trộn ứng với nhiệt độ của nó là $44\,^\circ\text{C}$ là bao nhiêu lít/giây (làm tròn kết quả đến chữ số hàng phần mười)?

		\shortans[oly]{1{,}3}
		\loigiai{
			$J_2=3J_1$ nên lưu lượng lớn nhất khi ống 2 mở cực đại: $J_2=1{,}00$~lít/s, $J_1=\dfrac{1}{3}$~lít/s.\\

	\begin{ex}%[VL12-C2-VD]
		\label{NgheAn-LienTruong-Lan5-III3}\textit{Dùng thông tin sau cho câu~\ref{NgheAn-LienTruong-Lan5-III3} và câu~\ref{NgheAn-LienTruong-Lan5-III4}:} Thể tích khí lưu thông (thể tích khí hít vào hoặc thở ra khi hít thở bình thường) của một người là $0{,}50$~lít. Không khí hít vào có $78{,}06\%$ thể tích là nitrogen, $20{,}95\%$ thể tích là oxygen và một tỉ lệ nhỏ các loại khí khác. Khí thở ra có $16{,}00\%$ thể tích là oxygen. Coi áp suất khí khi hít vào, thở ra đều bằng áp suất khí quyển, bằng $1{,}01\cdot 10^5$~Pa, nhiệt độ khí là $37\,^\circ\text{C}$.\\
		Số mol khí oxygen đã được máu hấp thụ trong mỗi lần trao đổi khí là $x\cdot 10^{-4}$~mol. Tìm $x$ (làm tròn kết quả đến chữ số hàng phần mười).
		\par
		\shortans[oly]{9{,}7}
		\loigiai{

			$n_{\text{O}_2}=(0{,}2095-0{,}1600)\,n\approx 9{,}70\cdot 10^{-4}~\text{mol}\Rightarrow x\approx 9{,}7$.
		}
	\end{ex}
