		-- chu kì bán rã của U-238: $T_1=4{,}47\cdot 10^9$ năm;\\
		-- chu kì bán rã của Th-234: $T_2=24{,}1$ ngày;\\

		\choiceTFt
		{\True Khối lượng U-238 có trong mẫu quặng là $1{,}99$~kg}

				$N_2=N_1\dfrac{T_2}{T_1}=5{,}03\cdot 10^{24}\cdot\dfrac{24{,}1}{4{,}47\cdot 10^9\cdot 365}\approx 7{,}42\cdot 10^{13}$ hạt.
