		Người ta dùng ấm điện có công suất $1000$~W để đun một lượng nước có khối lượng $m$ từ $20\,^\circ\text{C}$ lên tới $100\,^\circ\text{C}$. Thời gian đun là $7$ phút; coi hiệu suất là $100\,\%$, bỏ qua lượng nước hóa hơi; biết nhiệt dung riêng của nước là $4200$~J/(kg.K). Giá trị $m$ bằng bao nhiêu kg (kết quả làm tròn đến chữ số hàng phần trăm)?
		\par
		\shortans[oly]{1{,}25}
		\loigiai{
			$\mathcal{P}t=mc\Delta t\Rightarrow m=\dfrac{\mathcal{P}t}{c\Delta t}=\dfrac{1000\cdot 420}{4200\cdot 80}=1{,}25$~kg.
		}
	\end{ex}

	\NeedspaceCau{5.6\baselineskip}
	\begin{ex}%[VL12-C2-VD]
		Một bình có dung tích $10$ lít chứa một lượng khí hydrogen có khối lượng mol là $2$~g/mol bị nén ở áp suất $5\cdot 10^6$~Pa và nhiệt độ $7\,^\circ\text{C}$. Khi nung nóng bình, do bình hở nên có một phần khí thoát ra; phần khí còn lại có nhiệt độ $17\,^\circ\text{C}$ và vẫn dưới áp suất như cũ. Khối lượng khí đã thoát ra khỏi bình là bao nhiêu gam (kết quả làm tròn đến chữ số hàng phần chục)?
		\par
		\shortans[oly]{1{,}5}
		\loigiai{
			$m=\dfrac{pVM}{RT}$: $m_1=\dfrac{5\cdot 10^6\cdot 10^{-2}\cdot 2}{8{,}31\cdot 280}\approx 42{,}98$~g; $m_2=\dfrac{5\cdot 10^6\cdot 10^{-2}\cdot 2}{8{,}31\cdot 290}\approx 41{,}50$~g.\\
			$\Delta m=m_1-m_2\approx 1{,}48~\text{g}\approx 1{,}5$~g.
		}
	\end{ex}

	\NeedspaceCau{3.6\baselineskip}
	\begin{ex}%[VL12-C1-TH]
		Một vật nhận nhiệt lượng $200$~J, nó dãn nở và thực hiện công $40$~J. Độ biến thiên nội năng của vật đó bằng bao nhiêu J?
		\par
		\shortans[oly]{160}
		\loigiai{
			$\Delta U=A+Q=-40+200=160$~J.
		}
	\end{ex}

	\NeedspaceCau{10.6\baselineskip}
	\begin{ex}%[VL12-C1-VD]
		Trong thí nghiệm xác định nhiệt hóa hơi riêng của nước, người ta dùng một nguồn điện ổn định cung cấp công suất $500$~W để đun nước chứa trong một nhiệt lượng kế mở nắp. Để xác định khối lượng của nước còn lại trong nhiệt lượng kế, người ta đặt nhiệt lượng kế lên cân điện tử để đo rồi lấy kết quả hiển thị trên cân điện tử trừ đi khối lượng của nhiệt lượng kế. Sau khi đun được một thời gian thì nước bắt đầu sôi, khối lượng nước còn lại trong nhiệt lượng kế được xác định là $165$~g. Đun tiếp cho nước sôi trong $5$ phút nữa thì khối lượng nước còn lại trong nhiệt lượng kế là $100$~g. Coi hiệu suất của quá trình đun là $100\,\%$. Nhiệt hóa hơi riêng của nước được xác định thông qua thí nghiệm này là $x\cdot 10^6$~J/kg. Tìm $x$ (kết quả làm tròn đến chữ số hàng phần chục).
		\par
		\shortans[oly]{2{,}3}
		\loigiai{
			$L=\dfrac{\mathcal{P}t}{\Delta m}=\dfrac{500\cdot 300}{0{,}165-0{,}100}\approx 2{,}31\cdot 10^6~\text{J/kg}\Rightarrow x\approx 2{,}3$.
		}
	\end{ex}

	\NeedspaceCau{5.6\baselineskip}
	\begin{ex}%[VL12-C2-TH]
		Khi thở ra, dung tích của phổi là $2{,}40$ lít và áp suất của không khí trong phổi là $101{,}70\cdot 10^3$~Pa. Cho biết khi hít vào, áp suất này có giá trị $101{,}01\cdot 10^3$~Pa. Nhiệt độ của cơ thể người không thay đổi. Dung tích của phổi khi hít vào là bao nhiêu lít (kết quả làm tròn đến chữ số hàng phần trăm)?
		\par
		\shortans[oly]{2{,}42}
		\loigiai{
			Đẳng nhiệt: $V_2=\dfrac{p_1V_1}{p_2}=\dfrac{101{,}70\cdot 2{,}40}{101{,}01}\approx 2{,}42$ lít.
		}
	\end{ex}

	\NeedspaceCau{4.6\baselineskip}
	\begin{ex}%[VL12-C2-TH]
		Coi không khí là một khí đồng nhất có khối lượng mol $29$~g/mol. Nếu không khí ở nhiệt độ $17\,^\circ\text{C}$ thì tốc độ căn quân phương của phân tử khí (căn bậc hai của trung bình các bình phương tốc độ) bằng bao nhiêu m/s (kết quả làm tròn đến chữ số hàng đơn vị)?
		\par
		\shortans[oly]{499}
		\loigiai{
			$v_{\text{cqp}}=\sqrt{\dfrac{3RT}{M}}=\sqrt{\dfrac{3\cdot 8{,}31\cdot 290}{29\cdot 10^{-3}}}\approx 499$~m/s.
		}
	\end{ex}
