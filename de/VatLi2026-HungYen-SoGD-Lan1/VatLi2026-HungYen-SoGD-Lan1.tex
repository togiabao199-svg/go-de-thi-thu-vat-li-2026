Cho áp suất khí quyển là $10^5$~Pa. Cho hằng số khí lí tưởng $R=8{,}31$~J/(mol.K), gia tốc rơi tự do $g=10~\text{m/s}^2$, bỏ qua ma sát giữa các pit-tông với thành xilanh. Cần phải làm nóng khí trong xilanh thêm bao nhiêu $^\circ$C để các pit-tông dịch chuyển lên trên một đoạn $4$~cm (làm tròn kết quả đến chữ số hàng phần trăm)?
		\end{minipage}\hfill
		\begin{minipage}[c]{0.22\linewidth}
			\centering
			\begin{tikzpicture}[scale=0.62,line join=round,line cap=round,font=\footnotesize]
				% khí giữa hai pit-tông
				\fill[VLxanh!10] (0,1.95)--(0,4.6)--(2.5,4.6)--(2.5,1.95)--(1.6,1.95)--(1.6,0.6)--(0.9,0.6)--(0.9,1.95)--cycle;
				% pit-tông trên, dưới, thanh nối
				\shade[left color=gray!20,right color=gray!45,draw=black!70] (1.2,0.6) rectangle (1.3,4.6);
				\shade[top color=gray!25,bottom color=gray!65,draw=black] (0,4.6) rectangle (2.5,4.95);
				\shade[top color=gray!25,bottom color=gray!65,draw=black] (0.9,0.3) rectangle (1.6,0.6);
				% thành xilanh
				\draw[thick] (0,5.5)--(0,1.95)--(0.9,1.95)--(0.9,-0.3) (2.5,5.5)--(2.5,1.95)--(1.6,1.95)--(1.6,-0.3);
				\node[above,inner sep=2pt] at (1.25,5.5) {$m_1S_1$};
				\node[right,inner sep=3pt] at (1.6,0.45) {$m_2S_2$};
			\end{tikzpicture}
		\end{minipage}
		\par
		\shortans[oly]{0{,}67}
		\loigiai{
			Cân bằng hệ hai pit-tông và thanh nối: $pS_1+p_0S_2=p_0S_1+pS_2+Mg\Rightarrow p=p_0+\dfrac{Mg}{S_1-S_2}=10^5+\dfrac{40}{10^{-3}}=1{,}4\cdot 10^5$~Pa (không đổi khi pit-tông dịch chuyển chậm).\\
			$\Delta V=(S_1-S_2)h=10^{-3}\cdot 0{,}04=4\cdot 10^{-5}~\text{m}^3$.
			\[p\Delta V=nR\Delta T\Rightarrow \Delta T=\dfrac{1{,}4\cdot 10^5\cdot 4\cdot 10^{-5}}{1\cdot 8{,}31}\approx 0{,}67\,^\circ\text{C}.\]
		}
	\end{ex}

	\begin{ex}%[VL12-C2-TH]
		Khi xe đứng yên, khối không khí trong lốp xe ô tô con có nhiệt độ $28\,^\circ\text{C}$, áp suất $2{,}0685\cdot 10^5$~Pa. Khi lốp lăn trên đường thì khối không khí trong lốp có thể tích tăng thêm $2\%$ so với khi đứng yên, nhiệt độ bằng $48\,^\circ\text{C}$ và áp suất là $x\cdot 10^5$~Pa. Coi không khí trong lốp là lí tưởng. Giá trị của $x$ là bao nhiêu (làm tròn kết quả đến hàng phần trăm)?
		\par
		\shortans[oly]{2{,}16}
		\loigiai{
			$\dfrac{p_1V_1}{T_1}=\dfrac{p_2V_2}{T_2}\Rightarrow p_2=p_1\cdot\dfrac{T_2}{T_1}\cdot\dfrac{V_1}{V_2}=2{,}0685\cdot 10^5\cdot\dfrac{321}{301}\cdot\dfrac{1}{1{,}02}\approx 2{,}16\cdot 10^5$~Pa.
		}
	\end{ex}

	\NeedspaceCau{6.6\baselineskip}
	\begin{ex}%[VL12-C2-TH]
		Một nút chai bị kẹt ở cổ một chai thủy tinh chứa không khí có áp suất bằng áp suất khí quyển $10^5$~Pa và nhiệt độ $26\,^\circ\text{C}$. Người ta hơ nóng khí trong chai, nút bật ra nếu áp suất của khí trong chai lớn hơn hoặc bằng $1{,}2\cdot 10^5$~Pa. Coi khí trong chai là khí lí tưởng và không bị thoát ra ngoài khi hơ nóng, sự dãn nở vì nhiệt của chai không đáng kể. Nhiệt độ tối thiểu của khí để nút chai bật ra là bao nhiêu $^\circ$C (làm tròn kết quả đến chữ số hàng phần mười)?
		\par
		\shortans[oly]{85{,}8}
		\loigiai{
			Đẳng tích: $T_2=T_1\cdot\dfrac{p_2}{p_1}=299\cdot 1{,}2=358{,}8~\text{K}\Rightarrow t_2=85{,}8\,^\circ\text{C}$.
		}
	\end{ex}

	\NeedspaceCau{6.5\baselineskip}
	\begin{ex}%[VL12-C1-VD]
		Dẫn $0{,}5$~kg hơi nước ở nhiệt độ $100\,^\circ\text{C}$ từ một lò hơi vào một bình chứa $1$~kg nước đá ở $0\,^\circ\text{C}$. Cho biết nhiệt dung riêng của nước là $c=4200$~J/(kg.K), nhiệt hóa hơi riêng của nước là $L=2{,}3\cdot 10^6$~J/kg và nhiệt nóng chảy riêng của nước đá là $\lambda=3{,}4\cdot 10^5$~J/kg, bỏ qua sự hấp thụ nhiệt của bình chứa. Khi có cân bằng nhiệt, khối lượng nước ở trong bình khi đó là bao nhiêu kg (làm tròn đến $2$ chữ số sau dấu phẩy thập phân)?
		\par
		\shortans[oly]{1{,}33}
		\loigiai{
			Nhiệt lượng để nước đá tan hết và nóng lên tới $100\,^\circ\text{C}$: $Q_1=\lambda m_1+m_1c\cdot 100=3{,}4\cdot 10^5+4{,}2\cdot 10^5=7{,}6\cdot 10^5$~J.\\
			Hơi nước ngưng tụ hết tỏa $Lm_2=1{,}15\cdot 10^6~\text{J}>Q_1$ nên cân bằng ở $100\,^\circ\text{C}$, chỉ một phần hơi ngưng tụ:
			\[m=\dfrac{Q_1}{L}=\dfrac{7{,}6\cdot 10^5}{2{,}3\cdot 10^6}\approx 0{,}330~\text{kg}\Rightarrow m_{\text{nước}}=1+0{,}330\approx 1{,}33~\text{kg}.\]
		}
	\end{ex}
