		Một mol khí hydrogen ($\text{H}_2$) chứa trong một xilanh đậy kín bởi một pit-tông biến đổi chậm từ trạng thái (1) đến trạng thái (2) theo đồ thị (hình bên). Biết rằng ở trạng thái (1) khí có thể tích $30$~lít và áp suất $5$~atm, ở trạng thái (2) khí có thể tích là $10$~lít và áp suất $15$~atm. Tỉ số giữa tốc độ căn quân phương cực đại và cực tiểu của các phân tử khí trong quá trình trên là bao nhiêu (kết quả làm tròn đến chữ số hàng phần mười)?
		\begin{center}
			\begin{tikzpicture}[x=0.95cm,y=0.95cm,line join=round,line cap=round,font=\footnotesize]
				\draw[VLmui,thick] (0,0)--(3.6,0) node[above,inner sep=3pt] {$V$};
				\draw[VLmui,thick] (0,0)--(0,2.95) node[right,inner sep=2pt] {$p$};
				\node[below left,inner sep=2pt] at (0,0) {$O$};
				\draw[gray,dashed,thin] (0,2.4)--(1,2.4)--(1,0) (0,0.8)--(3,0.8)--(3,0);
				\draw[very thick,VLdo,VLgiua=0.55] (3,0.8)--(1,2.4);
				\foreach \p in {(3,0.8),(1,2.4)}{\filldraw[fill=white,draw=black,thick] \p circle (2pt);}
				\node[right,inner sep=3pt] at (3,0.8) {(1)};
				\node[above,inner sep=3pt] at (1,2.4) {(2)};
				\node[left,inner sep=2pt] at (0,2.4) {$p_2$};
				\node[left,inner sep=2pt] at (0,0.8) {$p_1$};
				\node[below,inner sep=2pt] at (1,0) {$V_2$};
				\node[below,inner sep=2pt] at (3,0) {$V_1$};
			\end{tikzpicture}
		\end{center}
		\shortans[oly]{1{,}2}
		\loigiai{
			Phương trình đoạn thẳng (1) $\to$ (2): $p=20-0{,}5V$ ($p$ tính bằng atm, $V$ bằng lít), $nRT=pV=20V-0{,}5V^2$.\\
			$pV$ cực đại tại $V=20$~lít: $(pV)_{\max}=200$; cực tiểu tại hai đầu: $(pV)_{\min}=150$.\\
			$v_{\text{rms}}=\sqrt{\dfrac{3RT}{\mu}}\Rightarrow\dfrac{v_{\max}}{v_{\min}}=\sqrt{\dfrac{T_{\max}}{T_{\min}}}=\sqrt{\dfrac{200}{150}}\approx 1{,}2$.
		}
