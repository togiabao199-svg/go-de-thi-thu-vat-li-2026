	\begin{ex}%[VL12-C3-TH]
		Một khung dây dẫn tròn có diện tích $60~\text{cm}^2$, đặt trong từ trường đều. Góc giữa $\vv{B}$ và mặt phẳng khung dây là $30^\circ$. Trong thời gian $0{,}01$~s từ trường tăng đều từ $0$ lên $0{,}02$~T thì suất điện động cảm ứng xuất hiện trong vòng dây có độ lớn là $0{,}6$~V. Khung dây trên gồm bao nhiêu vòng?
		\par
		\shortans[oly]{100}
		\loigiai{
			Góc giữa $\vv{B}$ và pháp tuyến là $60^\circ$: $|e|=NS\cos 60^\circ\dfrac{\Delta B}{\Delta t}$.
			\[N=\dfrac{|e|\,\Delta t}{S\cos 60^\circ\,\Delta B}=\dfrac{0{,}6\cdot 0{,}01}{60\cdot 10^{-4}\cdot 0{,}5\cdot 0{,}02}=100.\]
		}
	\end{ex}

	\NeedspaceCau{5.6\baselineskip}
	\begin{ex}%[VL12-C2-TH]
		Một bọt khí nổi từ đáy giếng sâu $8$~m lên mặt nước. Biết áp suất khí quyển là $1{,}013\cdot 10^5$~Pa, khối lượng riêng của nước giếng là $1000~\text{kg/m}^3$ và nhiệt độ của nước giếng không thay đổi theo độ sâu. Lấy $g=10~\text{m/s}^2$. Khi lên tới mặt nước, thể tích của bọt khí tăng bao nhiêu lần (làm tròn kết quả đến chữ số hàng phần trăm)?
		\par
		\shortans[oly]{1{,}79}
		\loigiai{
			Ở đáy giếng: $p_1=p_0+\rho gh=1{,}013\cdot 10^5+1000\cdot 10\cdot 8=1{,}813\cdot 10^5$~Pa.\\
			Đẳng nhiệt: $\dfrac{V_2}{V_1}=\dfrac{p_1}{p_0}=\dfrac{1{,}813}{1{,}013}\approx 1{,}79$.
		}
	\end{ex}

	\NeedspaceCau{6.6\baselineskip}
	\begin{ex}%[VL12-C1-VD]
		Bạn Mạnh đun sôi $1$~kg nước từ $25\,^\circ\text{C}$. Sau khi sôi một thời gian, Mạnh ngắt điện và rót lượng nước sôi còn lại vào phích có sẵn $200$~g nước ở $60\,^\circ\text{C}$. Nhiệt độ cân bằng trong phích là $92\,^\circ\text{C}$. Biết nhiệt dung riêng và nhiệt hóa hơi riêng của nước lần lượt là $4200$~J/(kg.K) và $2{,}3\cdot 10^6$~J/kg. Bỏ qua sự trao đổi nhiệt của nước với ruột phích. Nhiệt lượng đã cung cấp cho nước trong suốt quá trình đun là bao nhiêu kJ?
		\par
		\shortans[oly]{775}
		\loigiai{
			Cân bằng nhiệt trong phích: $m\,c\,(100-92)=0{,}2\,c\,(92-60)\Rightarrow m=0{,}8$~kg; lượng nước đã hóa hơi $1-0{,}8=0{,}2$~kg.
			\[Q=1\cdot 4200\cdot(100-25)+0{,}2\cdot 2{,}3\cdot 10^6=775\,000~\text{J}=775~\text{kJ}.\]
		}
	\end{ex}

	\NeedspaceCau{16.5\baselineskip}
	\begin{ex}%[VL12-C3-VD]
		Hai thanh dẫn không điện trở Ax và Qy, đặt song song, thẳng đứng trong không gian có từ trường đều. Vectơ cảm ứng từ nằm ngang và vuông góc với mặt phẳng chứa hai thanh dẫn (hình bên), độ lớn $0{,}50$~T. Nối hai đầu A, Q bên trên của hai thanh dẫn bằng một sợi dây không điện trở. Thanh kim loại MN có khối lượng $0{,}10$~kg, dài $0{,}40$~m, điện trở $R=0{,}25~\Omega$ được thả rơi không vận tốc đầu, trong quá trình rơi luôn tiếp xúc với hai thanh dẫn Ax và Qy, chuyển động theo phương song song với hai thanh. Chiều dài các thanh dẫn Ax và Qy đủ dài, bỏ qua ma sát và lực cản, lấy $g=10~\text{m/s}^2$. Tốc độ lớn nhất mà thanh kim loại MN đạt được bằng bao nhiêu m/s?
		\begin{center}
			\begin{tikzpicture}[scale=0.85,line join=round,line cap=round,font=\footnotesize]
				\fill[VLxanh!8] (0,0) rectangle (2.4,2.7);
				\draw[thick] (0,0)--(0,2.7)--(2.4,2.7)--(2.4,0);
				\shade[top color=VLdong!50!white,bottom color=VLdong!85!black] (0,1.6) rectangle (2.4,1.72);
				\draw[VLdong!40!black,thin] (0,1.6) rectangle (2.4,1.72);

				\node[above right,inner sep=1.5pt] at (2.4,2.7) {Q};
				\node[left,inner sep=2pt] at (0,1.66) {M};
				\node[right,inner sep=2pt] at (2.4,1.66) {N};
				\node[below left,inner sep=1.5pt] at (0,0) {x};
				\node[below right,inner sep=1.5pt] at (2.4,0) {y};
				\draw[VLxanh!85!black,semithick] (1.2,0.9) circle (0.17) ++(-0.12,-0.12)--++(0.24,0.24) (1.2,0.9)++(-0.12,0.12)--++(0.24,-0.24);

			\[v_{\max}=\dfrac{mgR}{B^2l^2}=\dfrac{0{,}10\cdot 10\cdot 0{,}25}{0{,}50^2\cdot 0{,}40^2}=6{,}25~\text{m/s}.\]
		}
