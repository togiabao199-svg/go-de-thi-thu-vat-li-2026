Biết: không khí bên ngoài có nhiệt độ $27\,^\circ\text{C}$, áp suất $10^5$~Pa; trước khi bơm, trong khí cầu chưa có khí; phía dưới của khí cầu có chế tạo một lỗ thông hơi với bên ngoài, lỗ này cũng là lỗ cấp nhiệt khi nung nóng không khí bên trong khí cầu; khối lượng mol của không khí là $29$~g/mol.

		Nhiệt lượng mà $1{,}5$~lít nước nhận được để tăng từ $20\,^\circ\text{C}$ đến sôi ở $100\,^\circ\text{C}$ là bao nhiêu kJ (làm tròn kết quả đến hàng đơn vị)?
		\par
		\shortans[oly]{504}
		\loigiai{
			$Q_1=m_1c_1\Delta t=1{,}5\cdot 4200\cdot 80=504\,000~\text{J}=504$~kJ.
		}
	\end{ex}

	\NeedspaceCau{3.5\baselineskip}
	\begin{ex}%[VL12-C1-VD]
		\label{PhuTho-SoGD-Lan1-III4}
		Công suất trung bình của bếp điện là bao nhiêu oát (W) (làm tròn kết quả đến hàng đơn vị)?
		\par
		\shortans[oly]{777}
		\loigiai{
			Nhiệt lượng có ích: $Q=m_2c_2\Delta t+m_1c_1\Delta t+0{,}2m_1L$
			\[Q=0{,}6\cdot 880\cdot 80+504\,000+0{,}2\cdot 1{,}5\cdot 2{,}26\cdot 10^6=1\,224\,240~\text{J}.\]
			$P=\dfrac{Q}{Ht}=\dfrac{1\,224\,240}{0{,}75\cdot 35\cdot 60}\approx 777$~W.
		}
	\end{ex}

	\NeedspaceCau{11.4\baselineskip}
	\begin{ex}%[VL12-C2-VD]
		\label{PhuTho-SoGD-Lan1-III5}
		\begin{minipage}[t]{0.7\linewidth}
			\textit{Dùng thông tin sau cho câu~\ref{PhuTho-SoGD-Lan1-III5} và câu~\ref{PhuTho-SoGD-Lan1-III6}:} Một bình kín hình trụ, đặt thẳng đứng có chiều dài $L$ được chia thành hai phần nhờ pit-tông mỏng cách nhiệt có khối lượng $m=600$~g. Phần 1 chứa khí helium (He), phần 2 chứa khí hydrogen ($\text{H}_2$) có cùng khối lượng và ở cùng nhiệt độ $27\,^\circ\text{C}$. Pit-tông cân bằng và cách đáy dưới một đoạn $0{,}6L$. Tiết diện bình là $S=1~\text{dm}^2$, lấy $g=10~\text{m/s}^2$. Khối lượng mol của He và $\text{H}_2$ lần lượt là $4$~g/mol, $2$~g/mol. Bỏ qua ma sát giữa pit-tông và xilanh.
		\end{minipage}\hfill
		\begin{minipage}[t]{0.18\linewidth}
			\centering
			\begin{tikzpicture}[baseline={([yshift=-1.2ex]current bounding box.north)},line join=round,line cap=round,font=\footnotesize]
				\fill[VLxanh!8] (0,0) rectangle (1,2.4);
				\shade[top color=gray!30,bottom color=gray!65,draw=black!70] (0,1.44) rectangle (1,1.62);
				\draw[thick] (0,0) rectangle (1,2.4);
				\node at (0.5,2.02) {1};
				\node at (0.5,0.72) {2};
				\draw[{Stealth[length=4pt]}-{Stealth[length=4pt]},semithick] (-0.25,0)--(-0.25,2.4);
				\node[left,inner sep=2pt] at (-0.25,1.2) {$L$};
				\draw[{Stealth[length=4pt]}-{Stealth[length=4pt]},semithick] (1.25,0)--(1.25,1.44);
				\node[right,inner sep=2pt] at (1.25,0.72) {$0{,}6L$};
			\end{tikzpicture}
		\end{minipage}
		\par\noindent
		Áp suất khí trong phần 2 khi nhiệt độ $27\,^\circ\text{C}$ bằng bao nhiêu kPa (làm tròn kết quả đến chữ số hàng phần mười)?
		\par
		\shortans[oly]{2{,}4}
		\loigiai{
			$p_1=\dfrac{m_{\text{k}}RT}{4\cdot 0{,}4LS}$, $p_2=\dfrac{m_{\text{k}}RT}{2\cdot 0{,}6LS}\Rightarrow p_1=\dfrac{3}{4}p_2$.\\
			Cân bằng pit-tông: $p_2S=p_1S+mg\Rightarrow \dfrac{1}{4}p_2=\dfrac{mg}{S}=\dfrac{0{,}6\cdot 10}{10^{-2}}=600~\text{Pa}\Rightarrow p_2=2400~\text{Pa}=2{,}4$~kPa.
		}
	\end{ex}

	\NeedspaceCau{4.2\baselineskip}
	\begin{ex}%[VL12-C2-VD]
		\label{PhuTho-SoGD-Lan1-III6}
		Giữ nhiệt độ ở phần 2 không đổi, nung nóng phần 1 đến nhiệt độ $600$~K thì pit-tông cách đáy dưới khoảng là $h$. Tìm tỉ số $\dfrac{h}{L}$ (làm tròn kết quả đến chữ số hàng phần trăm).
		\par
		\shortans[oly]{0{,}45}
		\loigiai{
			Đặt $x=\dfrac{h}{L}$; $p_1=1800$~Pa, $p_2=2400$~Pa.\\
			Phần 2 đẳng nhiệt: $p_2'=\dfrac{2400\cdot 0{,}6}{x}$; phần 1: $p_1'=1800\cdot\dfrac{0{,}4}{1-x}\cdot\dfrac{600}{300}=\dfrac{1440}{1-x}$.
			\[p_2'-p_1'=600\Rightarrow \dfrac{1440}{x}-\dfrac{1440}{1-x}=600\Rightarrow x^2-5{,}8x+2{,}4=0\Rightarrow x\approx 0{,}45.\]
		}
	\end{ex}
