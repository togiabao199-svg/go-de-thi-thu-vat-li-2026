	\begin{ex}%[VL12-C4-VD]
		``Mặt Trời nhân tạo'' là thuật ngữ chỉ các lò phản ứng nhiệt hạch mô phỏng quá trình tạo năng lượng của Mặt Trời thông qua phản ứng tổng hợp hạt nhân, đốt nóng plasma lên hơn $100$ triệu $^\circ$C để tạo ra nguồn năng lượng sạch vô tận.\\
		Xét phản ứng tổng hợp hạt nhân ${}^2_1\text{D}+{}^2_1\text{D}\to{}^3_2\text{He}+{}^1_0\text{n}$. Cho biết phản ứng này tỏa năng lượng khoảng $3{,}1671$~MeV. Nước trong tự nhiên chứa $0{,}015\%$ nước nặng D$_2$O; khối lượng mol của D$_2$O bằng $20$~g/mol. Nếu dùng toàn bộ ${}^2_1\text{D}$ có trong $1$~kg nước tự nhiên để làm nhiên liệu cho phản ứng trên thì năng lượng thu được là bao nhiêu MJ (làm tròn kết quả đến chữ số hàng đơn vị)?
		\par
		\shortans[oly]{2288}
		\loigiai{
			Khối lượng D$_2$O: $m=1000\cdot 0{,}015\%=0{,}15$~g; $n=\dfrac{0{,}15}{20}=7{,}5\cdot 10^{-3}$~mol.\\
			Mỗi phân tử D$_2$O có $2$ hạt D, mỗi phản ứng dùng $2$ hạt D nên số phản ứng bằng số phân tử D$_2$O:
			\[E=7{,}5\cdot 10^{-3}\cdot 6{,}02\cdot 10^{23}\cdot 3{,}1671\cdot 1{,}6\cdot 10^{-13}\approx 2{,}288\cdot 10^9~\text{J}\approx 2288~\text{MJ}.\]
		}
	\end{ex}

	\NeedspaceCau{4.6\baselineskip}
	\begin{ex}%[VL12-C2-VD]
		Một bọt khí có bán kính $4$~mm đi từ đáy hồ đi lên mặt hồ. Nhiệt độ và áp suất ở đáy hồ là $7\,^\circ\text{C}$ và $2$~atm, ở mặt hồ là $27\,^\circ\text{C}$ và $1$~atm. Bán kính bọt khí ở sát mặt hồ là bao nhiêu mm (làm tròn kết quả đến chữ số hàng phần trăm)?
		\par
		\shortans[oly]{5{,}16}
		\loigiai{
			$\dfrac{p_1V_1}{T_1}=\dfrac{p_2V_2}{T_2}\Rightarrow \dfrac{V_2}{V_1}=\dfrac{p_1T_2}{p_2T_1}=\dfrac{2\cdot 300}{1\cdot 280}=\dfrac{15}{7}$.\\
			$V\sim r^3\Rightarrow r_2=r_1\sqrt[3]{\dfrac{15}{7}}=4\sqrt[3]{\dfrac{15}{7}}\approx 5{,}16$~mm.
		}
	\end{ex}

	\NeedspaceCau{4.6\baselineskip}
	\begin{ex}%[VL12-C2-TH]
		Một quả bóng đá được bơm căng từ nơi có nhiệt độ $2\,^\circ\text{C}$ và sau đó mang đến nơi có nhiệt độ $35\,^\circ\text{C}$. Xem thể tích quả bóng không thay đổi. Độ tăng áp suất không khí trong quả bóng là bao nhiêu phần trăm (làm tròn kết quả đến chữ số hàng đơn vị)?

		}
	\end{ex}

	\begin{ex}%[VL12-C3-VD]
		Phần ứng của một máy phát điện xoay chiều một pha có $200$ vòng dây. Từ thông qua mỗi vòng dây có giá trị cực đại là $2$~mWb và biến thiên điều hòa với tần số $50$~Hz. Tải tiêu thụ là điện trở thuần $R=1000~\Omega$ được nối vào hai cực của máy phát. Nhiệt lượng tỏa ra trên $R$ trong thời gian $1$ giờ là bao nhiêu kJ (làm tròn kết quả đến chữ số hàng phần mười)?

	\begin{ex}%[VL12-C4-VD]
		Đồng vị ${}^{238}\text{U}$ qua một chuỗi phân rã phóng xạ $\alpha$ và $\beta$ biến thành hạt nhân bền ${}^{206}\text{Pb}$. Biết chu kì bán rã của ${}^{238}\text{U}$ là $4{,}47\cdot 10^9$ năm. Một khối đá được phát hiện chứa $46{,}97$~mg ${}^{238}\text{U}$ và $23{,}15$~mg ${}^{206}\text{Pb}$. Giả sử khối đá khi mới hình thành không chứa nguyên tố chì và tất cả lượng chì có mặt trong đó đều là sản phẩm phân rã của ${}^{238}\text{U}$. Khối lượng mol của ${}^{238}\text{U}$ và ${}^{206}\text{Pb}$ lần lượt là $238$~g/mol và $206$~g/mol. Tuổi của khối đá đó là bao nhiêu tỉ năm (làm tròn kết quả đến chữ số hàng phần mười)?
		\par
		\shortans[oly]{2{,}9}
		\loigiai{
			Mỗi hạt nhân ${}^{238}\text{U}$ phân rã tạo một hạt nhân ${}^{206}\text{Pb}$:
			\[\dfrac{N_{\text{Pb}}}{N_{\text{U}}}=\dfrac{23{,}15/206}{46{,}97/238}\approx 0{,}5694=2^{\frac{t}{T}}-1.\]
			$t=T\cdot\dfrac{\ln(1+0{,}5694)}{\ln 2}=4{,}47\cdot\dfrac{\ln 1{,}5694}{0{,}693}\approx 2{,}9$ tỉ năm.
		}
	\end{ex}

		Để có một cốc nước trà giải khát, người ta bỏ $240$~g nước đá ở $0\,^\circ\text{C}$ vào một cốc chứa $210$~g nước trà ở $67{,}5\,^\circ\text{C}$. Cho nhiệt nóng chảy riêng của nước đá bằng $3{,}36\cdot 10^5$~J/kg, nhiệt dung riêng của nước trà bằng $4200$~J/(kg.K). Coi rằng chỉ có sự trao đổi nhiệt giữa nước trà và nước đá. Khi vừa xảy ra sự cân bằng nhiệt thì khối lượng nước đá chưa tan trong cốc là bao nhiêu gam (làm tròn kết quả đến chữ số hàng phần mười)?

			Nước trà hạ nhiệt về $0\,^\circ\text{C}$ tỏa ra: $Q=mc\Delta t=0{,}21\cdot 4200\cdot 67{,}5=59\,535$~J $<\lambda\cdot 0{,}24=80\,640$~J nên nước đá chỉ tan một phần, cân bằng ở $0\,^\circ\text{C}$.\\

		}
	\end{ex}
