Thể rắn nóng từ $-20\,^\circ\text{C}$ lên $-10\,^\circ\text{C}$ trong $1$ phút; nóng chảy ở $-10\,^\circ\text{C}$ từ phút thứ $1$ đến phút thứ $13$; thể lỏng nóng từ $-10\,^\circ\text{C}$ lên $30\,^\circ\text{C}$ trong $11$ phút rồi sôi ở $30\,^\circ\text{C}$.

				\itemch \textbf{Sai.} Nhiệt độ sôi là $30\,^\circ\text{C}$; $-10\,^\circ\text{C}$ là nhiệt độ nóng chảy.

	\subsection*{\indam{PHẦN III. Câu trắc nghiệm trả lời ngắn. Thí sinh trả lời từ câu 1 đến câu 6.}}

	\begin{ex}%[VL12-C1-NB]
		Khi làm thí nghiệm, một học sinh đã đo được nhiệt độ của một cốc nước ấm là $55\,^\circ\text{C}$. Trong thang Kelvin thì nhiệt độ này ứng với bao nhiêu K?
		\par
		\shortans[oly]{328}
		\loigiai{
			$T=t+273=55+273=328$~K.
		}
	\end{ex}

	\NeedspaceCau{5.6\baselineskip}
	\begin{ex}%[VL12-C2-VD]
		Một quả bóng chuyền khi thi đấu có thể tích $4{,}85$ lít và áp suất $1{,}3$~atm. Sử dụng một cái bơm tay để bơm không khí vào bóng, mỗi lần bơm đưa được $0{,}63$ lít không khí ở áp suất $1$~atm vào bóng. Bơm chậm để nhiệt độ không đổi. Ban đầu trong bóng không có không khí. Số lần bơm bóng là bao nhiêu (kết quả làm tròn đến chữ số hàng đơn vị)?
		\par
		\shortans[oly]{10}
		\loigiai{
			Đẳng nhiệt cho toàn bộ lượng khí bơm vào: $N\cdot p_0V_0=pV$.
			\[N=\dfrac{pV}{p_0V_0}=\dfrac{1{,}3\cdot 4{,}85}{1\cdot 0{,}63}\approx 10{,}01\Rightarrow N=10.\]
		}
	\end{ex}

	\NeedspaceCau{5.6\baselineskip}
	\begin{ex}%[VL12-C2-VD]
		Một người mỗi nhịp thở hít vào $0{,}50$ lít không khí ở áp suất $10^5$~Pa và nhiệt độ $27\,^\circ\text{C}$. Biết khối lượng mol của không khí là $29$~g/mol, không khí có $21{,}0\,\%$ số phân tử là oxygen. Mỗi nhịp thở người này hít vào số phân tử oxygen là $x\cdot 10^{21}$. Giá trị của $x$ là bao nhiêu (kết quả làm tròn đến chữ số hàng phần mười)?
		\par
		\shortans[oly]{2{,}5}
		\loigiai{
			$n=\dfrac{pV}{RT}=\dfrac{10^5\cdot 0{,}5\cdot 10^{-3}}{8{,}31\cdot 300}\approx 2{,}006\cdot 10^{-2}$~mol.\\
			$N_{\text{O}_2}=0{,}21nN_A=0{,}21\cdot 2{,}006\cdot 10^{-2}\cdot 6{,}02\cdot 10^{23}\approx 2{,}5\cdot 10^{21}\Rightarrow x\approx 2{,}5$.
		}
	\end{ex}

	\NeedspaceCau{4.4\baselineskip}
	\begin{ex}%[VL12-C1-TH]
		Đun $500$~g nước đến $100\,^\circ\text{C}$ thì nước sôi. Biết nước có nhiệt hóa hơi riêng là $2{,}26\cdot 10^6$~J/kg. Để hóa hơi hoàn toàn khối nước này ở $100\,^\circ\text{C}$ thì nhiệt lượng cần cung cấp là $x\cdot 10^6$~J. Giá trị của $x$ là bao nhiêu?
		\par
		\shortans[oly]{1{,}13}
		\loigiai{
			$Q=Lm=2{,}26\cdot 10^6\cdot 0{,}5=1{,}13\cdot 10^6~\text{J}\Rightarrow x=1{,}13$.
		}
	\end{ex}

	\NeedspaceCau{4.6\baselineskip}
	\begin{ex}%[VL12-C2-TH]
		Vành nhật hoa của Mặt Trời có nhiệt độ là $6\cdot 10^5$~K. Lấy hằng số Boltzmann $k=1{,}38\cdot 10^{-23}$~J/K. Động năng tịnh tiến trung bình của mỗi nguyên tử helium trong vùng này là $x\cdot 10^{-18}$~J. Giá trị của $x$ là bao nhiêu (kết quả làm tròn đến chữ số hàng phần mười)?
		\par
		\shortans[oly]{12{,}4}
		\loigiai{
			$\overline{E}_{\text{đ}}=\dfrac{3}{2}kT=\dfrac{3}{2}\cdot 1{,}38\cdot 10^{-23}\cdot 6\cdot 10^5\approx 12{,}4\cdot 10^{-18}~\text{J}\Rightarrow x\approx 12{,}4$.
		}
	\end{ex}

	\NeedspaceCau{4.7\baselineskip}
	\begin{ex}%[VL12-C2-VD]
		Ở mặt hồ, áp suất khí quyển là $10^5$~Pa. Một bọt khí ở đáy hồ có độ sâu $5$~m nổi lên mặt nước thì thể tích của bọt khí tăng lên bao nhiêu lần? Giả sử nhiệt độ ở đáy hồ và mặt hồ là như nhau, khối lượng riêng của nước là $10^3~\text{kg/m}^3$, lấy $g=10~\text{m/s}^2$.
		\par
		\shortans[oly]{1{,}5}
		\loigiai{
			Áp suất ở đáy hồ: $p_1=p_0+\rho gh=10^5+10^3\cdot 10\cdot 5=1{,}5\cdot 10^5$~Pa.\\
			Đẳng nhiệt: $\dfrac{V_2}{V_1}=\dfrac{p_1}{p_0}=\dfrac{1{,}5\cdot 10^5}{10^5}=1{,}5$.
		}
	\end{ex}
