Biết rằng mỗi hạt nhân ${}^{235}_{\phantom{2}92}\text{U}$ phân hạch tỏa ra năng lượng trung bình là $200$~MeV. Hiệu suất của nhà máy điện là $30\%$. Nếu nhà máy hoạt động với công suất điện không đổi là $600$~MW thì trong một ngày đêm ($24$ giờ), khối lượng nhiên liệu uranium ${}^{235}_{\phantom{2}92}\text{U}$ thực tế đã tiêu thụ là bao nhiêu gam? Biết khối lượng mol của ${}^{235}_{\phantom{2}92}\text{U}$ là $235$~g/mol (làm tròn kết quả đến chữ số hàng đơn vị).
		\par
		\shortans[oly]{2108}
		\loigiai{
			Năng lượng phân hạch trong một ngày: $E=\dfrac{\mathscr{P}t}{H}=\dfrac{600\cdot 10^6\cdot 86\,400}{0{,}3}=1{,}728\cdot 10^{14}$~J.\\
			Số hạt nhân phân hạch: $N=\dfrac{1{,}728\cdot 10^{14}}{200\cdot 1{,}6\cdot 10^{-13}}=5{,}4\cdot 10^{24}$.\\
			$m=\dfrac{N}{N_A}\cdot M=\dfrac{5{,}4\cdot 10^{24}}{6{,}02\cdot 10^{23}}\cdot 235\approx 2108$~g.
		}
	\end{ex}
