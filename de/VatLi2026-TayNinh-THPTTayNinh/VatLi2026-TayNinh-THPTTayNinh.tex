				\draw[VLmui,thick,VLluc] (0,-0.2)--(0,-2.1);

	\begin{ex}%[VL12-C3-TH]
		Một khung dây dẫn kín, cứng, phẳng gồm $10$ vòng dây, có diện tích $25~\text{cm}^2$, đặt trong từ trường đều sao cho mặt phẳng khung vuông góc với các đường cảm ứng từ. Cảm ứng từ biến thiên theo thời gian như đồ thị hình vẽ. Độ lớn suất điện động cảm ứng xuất hiện trong khung trong khoảng thời gian từ $t=0$ đến $t=0{,}4$~s là bao nhiêu mV (làm tròn kết quả đến chữ số hàng phần trăm)?
		\begin{center}
			\begin{tikzpicture}[x=5cm,y=0.68cm,line join=round,line cap=round,font=\footnotesize]
				\draw[VLmui,thick] (0,0)--(0.56,0) node[below,inner sep=3pt] {$t$ (s)};
				\draw[VLmui,thick] (0,0)--(0,3.15) node[above,inner sep=2pt] {$B$ (mT)};
				\draw[very thick,VLxanh] (0,2.4)--(0.4,0);
				\node[left,inner sep=2pt] at (0,2.4) {$2{,}4$};
				\node[below,inner sep=2pt] at (0.4,0) {$0{,}4$};

		\end{center}

		\loigiai{
			$e=NS\left|\dfrac{\Delta B}{\Delta t}\right|=10\cdot 25\cdot 10^{-4}\cdot\dfrac{2{,}4\cdot 10^{-3}}{0{,}4}=1{,}5\cdot 10^{-4}~\text{V}=0{,}15$~mV.

	\NeedspaceCau{7.6\baselineskip}
	\begin{ex}%[VL12-C4-VD]
		Một nhà máy điện hạt nhân dùng nhiên liệu uranium $^{235}_{\phantom{2}92}\text{U}$. Biết công suất phát điện là $500$~MW và hiệu suất chuyển hóa năng lượng hạt nhân thành điện năng là $25\%$. Cho rằng khi một hạt nhân uranium $^{235}_{\phantom{2}92}\text{U}$ phân hạch thì tỏa ra năng lượng là $200$~MeV. Biết khối lượng mol của $^{235}_{\phantom{2}92}\text{U}$ là $235$~g/mol, $N_A=6{,}02\cdot 10^{23}$~hạt/mol. Nếu nhà máy hoạt động liên tục trong $365$ ngày thì lượng uranium $^{235}_{\phantom{2}92}\text{U}$ cần dùng là $x\cdot 10^2$~kg. Tìm $x$ (làm tròn kết quả đến chữ số hàng phần mười).
		\par
		\shortans[oly]{7{,}7}
		\loigiai{
			Năng lượng hạt nhân cần: $E=\dfrac{Pt}{H}=\dfrac{500\cdot 10^6\cdot 365\cdot 86\,400}{0{,}25}\approx 6{,}307\cdot 10^{16}$~J.\\
			Số hạt nhân phân hạch: $N=\dfrac{E}{200\cdot 1{,}6\cdot 10^{-13}}\approx 1{,}971\cdot 10^{27}$.
			\[m=\dfrac{N}{N_A}\cdot 235\approx 7{,}694\cdot 10^5~\text{g}\approx 7{,}7\cdot 10^2~\text{kg}\Rightarrow x\approx 7{,}7.\]
		}
	\end{ex}
