	\pgfmathsetmacro{\VLbuoc}{#1/(#3-1)}%

	\shade[top color=gray!60,bottom color=gray!80,middle color=gray!8] (-0.3,-0.72*#2) rectangle (#1+0.3,0.72*#2);
	\shade[left color=gray!25,right color=gray!60] (#1+0.3,0) ellipse ({0.2*#2} and {0.72*#2});
	\foreach \VLi in {1,...,#3}{%
		\draw[VLdong!80!black,line width=1.3pt] ({(\VLi-1)*\VLbuoc},#2) arc[start angle=90,end angle=-90,x radius=0.3*#2,y radius=#2];
