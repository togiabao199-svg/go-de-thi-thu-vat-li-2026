\def\VLledve#1#2#3#4#5#6#7{%
	\ifnum#1=1 \VLledngang{0.935}\fi \ifnum#2=1 \VLleddung{0.455}{0.5}{0.935}\fi \ifnum#3=1 \VLleddung{0.455}{0.065}{0.5}\fi

\newcommand{\VLsoled}[4]{%
	\xdef\VLledW{-0.16}%
	\foreach \VLc in {#4}{\if\VLc.\pgfmathparse{\VLledW+0.3}\else\pgfmathparse{\VLledW+0.68}\fi \xdef\VLledW{\pgfmathresult}}%
	\pgfmathparse{-0.5*\VLledW}\xdef\VLledX{\pgfmathresult}%
	\begin{scope}[shift={(#1,#2)},scale=#3,xslant=0.08,black!78]
		\foreach \VLc in {#4}{%
			\if\VLc.%
				\fill (\VLledX-0.06,0-0.5)--(\VLledX+0.07,0-0.5)--(\VLledX+0.07,0.13-0.5)--(\VLledX-0.06,0.13-0.5)--cycle;

	\end{scope}}
